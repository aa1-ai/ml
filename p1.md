```
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import LabelEncoder, StandardScaler
df = load_iris(as_frame=True).frame
df["species"] = df["target"].map({0:"setosa", 1:"versicolor", 2:"virginica"})
data = df.drop(columns="target").copy()
data.loc[[5,10,15,20], "sepal length (cm)"] = np.nan
data.loc[[25,30], "species"] = np.nan
print("Shape:", data.shape)
print("\nFirst 5 rows:\n", data.head())
print("\nMissing values:\n", data.isnull().sum())
before = data["sepal length (cm)"].copy()
data["sepal length (cm)"] = data["sepal length (cm)"].fillna(data["sepal length (cm)"].median())
data["species"] = data["species"].fillna(data["species"].mode()[0])
data["petal_category"] = pd.cut(data["petal length (cm)"], bins=[0,2,4,8], labels=["Small","Medium","Large"])
le = LabelEncoder()
data["species"] = le.fit_transform(data["species"])
print("\nSpecies Encoding:", dict(zip(le.classes_, le.transform(le.classes_))))
data = pd.get_dummies(data, columns=["petal_category"], dtype=int)
cols = data.columns[:4]
data[cols] = StandardScaler().fit_transform(data[cols])
print("\nMissing values after cleaning:", data.isnull().sum().sum())
print("\nFinal shape:", data.shape)
print("\nMean:\n", data[cols].mean().round(4))
print("\nStandard Deviation:\n", data[cols].std(ddof=0).round(4))
print("\nFinal dataset:\n", data.head().round(4))
fig, ax = plt.subplots(2,1,figsize=(8,8))
ax[0].hist(before.dropna(), bins=30, edgecolor="white")
ax[0].set_title("Before Imputation")
ax[1].hist(data["sepal length (cm)"], bins=30, edgecolor="white")
ax[1].set_title("After Imputation and Scaling")
plt.tight_layout()
plt.show()
```
